cmake_minimum_required(VERSION 3.22)
project(speech_llm_gpu C CXX)

set(CMAKE_CXX_STANDARD 17)
set(THIRD_PARTY ${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party)

set(SPEECH_ARM_ARCH "armv8.2-a+dotprod+fp16+i8mm" CACHE STRING "")
set(SPEECH_OPT_FLAGS "-O3 -DNDEBUG -ffunction-sections -fdata-sections")
set(CMAKE_C_FLAGS_RELEASE   "${SPEECH_OPT_FLAGS}" CACHE STRING "" FORCE)
set(CMAKE_CXX_FLAGS_RELEASE "${SPEECH_OPT_FLAGS}" CACHE STRING "" FORCE)
add_compile_options(-march=${SPEECH_ARM_ARCH})

# Chargeur ICD Khronos : uniquement pour l'édition de liens (libOpenCL.so exclue de l'APK).
set(OPENCL_ICD_LOADER_HEADERS_DIR ${THIRD_PARTY}/OpenCL-Headers CACHE PATH "" FORCE)
set(OPENCL_ICD_LOADER_BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(ENABLE_OPENCL_LAYERS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory(${THIRD_PARTY}/OpenCL-ICD-Loader OpenCL-ICD-Loader EXCLUDE_FROM_ALL)
# find_package(OpenCL) de ggml-opencl : pointer vers les en-têtes et la bibliothèque construite.
set(OpenCL_INCLUDE_DIR ${THIRD_PARTY}/OpenCL-Headers CACHE PATH "" FORCE)
set(OpenCL_LIBRARY $<TARGET_FILE:OpenCL> CACHE STRING "" FORCE)

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH ${SPEECH_ARM_ARCH} CACHE STRING "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_REPACK ON CACHE BOOL "" FORCE)
set(GGML_OPENCL ON CACHE BOOL "" FORCE)
set(GGML_OPENCL_USE_ADRENO_KERNELS ON CACHE BOOL "" FORCE)
set(GGML_OPENCL_EMBED_KERNELS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_ALL_WARNINGS OFF CACHE BOOL "" FORCE)
add_subdirectory(${THIRD_PARTY}/llama.cpp llama.cpp EXCLUDE_FROM_ALL)
add_dependencies(ggml-opencl OpenCL)

add_library(llama_jni_gpu SHARED ../../../../llm/src/main/cpp/llama_jni.cpp)
target_compile_definitions(llama_jni_gpu PRIVATE JNI_CLASS=com_karelisio_speech_llm_LlamaNativeGpu)
target_link_libraries(llama_jni_gpu PRIVATE llama android log)
target_link_options(llama_jni_gpu PRIVATE -Wl,--gc-sections -Wl,--exclude-libs,ALL)
